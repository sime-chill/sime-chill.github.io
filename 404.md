<script>
  (() => {
    const legacyPrefix = '/chipatlas/';
    if (window.location.pathname.startsWith(legacyPrefix)) {
      const destination =
        '/fama/' + window.location.pathname.slice(legacyPrefix.length);
      window.location.replace(
        destination + window.location.search + window.location.hash,
      );
    }
  })();
</script>

<div class="not-found-container">
  <h1 class="not-found-title">404</h1>

  <p><strong>Page is being found right now :(</strong></p>
</div>
